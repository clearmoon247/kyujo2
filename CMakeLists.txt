cmake_minimum_required(VERSION 3.20)
project(Kyujo2 VERSION 0.26.0 LANGUAGES C)
set(KYUJO2_REVISION "English r26" CACHE STRING "Revision label written to the log")
set(RECOMP_ROOT "${CMAKE_SOURCE_DIR}/.work/N64Recomp" CACHE PATH "N64Recomp source checkout")
set(KYUJO2_REGION "us" CACHE STRING "Cartridge region (us or jp)")
if(NOT KYUJO2_REGION MATCHES "^(us|jp)$")
    message(FATAL_ERROR "KYUJO2_REGION must be us or jp")
endif()
set(GAME_C "${CMAKE_SOURCE_DIR}/build/recompiled/${KYUJO2_REGION}")
file(GLOB GAME_UNITS CONFIGURE_DEPENDS "${GAME_C}/funcs_*.c")
if(NOT GAME_UNITS)
    message(FATAL_ERROR "Generate the ${KYUJO2_REGION} guest sources first")
endif()
set(GUEST_INCLUDE "${CMAKE_BINARY_DIR}/guest-include")
file(MAKE_DIRECTORY "${GUEST_INCLUDE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${RECOMP_ROOT}/include/recomp.h" "${CMAKE_SOURCE_DIR}/patches/recomp-rdram.patch")
execute_process(COMMAND patch --silent -o "${GUEST_INCLUDE}/recomp.tmp"
    "${RECOMP_ROOT}/include/recomp.h" "${CMAKE_SOURCE_DIR}/patches/recomp-rdram.patch"
    COMMAND_ERROR_IS_FATAL ANY)
configure_file("${GUEST_INCLUDE}/recomp.tmp" "${GUEST_INCLUDE}/recomp.h" COPYONLY)
add_library(stadium2_game STATIC ${GAME_UNITS} native/gsm.c native/asset_entry.c native/guest_link.c)
if(KYUJO2_REGION STREQUAL "jp")
    target_compile_definitions(stadium2_game PUBLIC KYUJO2_JP)
endif()
target_include_directories(stadium2_game PUBLIC "${GUEST_INCLUDE}" "${RECOMP_ROOT}/include" "${GAME_C}" native
    PRIVATE source/Kyujo-main/src/game_asm)
target_compile_options(stadium2_game PRIVATE -include guest_calls.h -ffunction-sections -fdata-sections
    -fno-strict-aliasing -fwrapv)
# The generated guest code runs every game frame and the Game Boy core's handlers chain through indirect tail calls,
# which need sibling-call optimization to keep the host stack flat. Optimize it in every build type.
option(KYUJO2_OPTIMIZE_GUEST "Compile the recompiled game code with -O2" ON)
if(KYUJO2_OPTIMIZE_GUEST)
    target_compile_options(stadium2_game PRIVATE -O2)
endif()

option(BUILD_RUNTIME_TRACE "Link the desktop runtime and exercise game startup" OFF)
if(BUILD_RUNTIME_TRACE)
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(RUNTIME_ROOT "${CMAKE_SOURCE_DIR}/.work/N64ModernRuntime" CACHE PATH "N64ModernRuntime checkout")
    if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
        add_compile_definitions(_DEFAULT_SOURCE)
        set(FMT_OS OFF CACHE BOOL "Disable unsupported host file descriptor wrappers" FORCE)
    endif()
    add_subdirectory("${RUNTIME_ROOT}" n64runtime EXCLUDE_FROM_ALL)
    set(rsp_sources build/rsp/${KYUJO2_REGION}-audio.cpp build/rsp/${KYUJO2_REGION}-jpeg.cpp)
    if(KYUJO2_REGION STREQUAL "us")
        list(APPEND rsp_sources build/rsp/us-gbcore.cpp build/rsp/us-gbcore-a.cpp build/rsp/us-gbcore-b.cpp)
    endif()
    set(host_sources native/trace.cpp native/controller.c native/desktop.cpp native/crash_log.cpp ${rsp_sources})
    if(ANDROID)
        add_library(Kyujo2 SHARED ${host_sources} native/android_touch.cpp)
        target_link_libraries(Kyujo2 PRIVATE android log)
        target_link_options(Kyujo2 PRIVATE "-Wl,-z,max-page-size=16384")
    else()
        add_executable(Kyujo2 ${host_sources})
    endif()
    if(KYUJO2_REGION STREQUAL "us")
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        set(KYUJO_REFERENCE_ROM "${CMAKE_SOURCE_DIR}/other_projects/Pokemon Stadium (USA).z64" CACHE FILEPATH "US cartridge for Kyujo menu artwork")
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/kyujo_menu_assets.inc"
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/kyujo-menu-assets.py"
                "${KYUJO_REFERENCE_ROM}" "${CMAKE_CURRENT_BINARY_DIR}/kyujo_menu_assets.inc"
            DEPENDS "${KYUJO_REFERENCE_ROM}" tools/kyujo-menu-assets.py VERBATIM)
        target_sources(Kyujo2 PRIVATE native/menu_canvas.c "${CMAKE_CURRENT_BINARY_DIR}/kyujo_menu_assets.inc")
        target_include_directories(Kyujo2 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        set(reference_platform "${CMAKE_SOURCE_DIR}/source/Kyujo-main/src/platform")
        set(menu_dir "${CMAKE_CURRENT_BINARY_DIR}/kyujo-menu")
        file(MAKE_DIRECTORY "${menu_dir}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
            "${CMAKE_SOURCE_DIR}/patches/kyujo2-shared-menu.patch" "${CMAKE_SOURCE_DIR}/patches/kyujo2-gbpak-rtc.patch"
            "${CMAKE_SOURCE_DIR}/patches/kyujo2-switch-replace.patch")
        foreach(unit host_menu host_paths host_settings host_gbpak)
            configure_file("${reference_platform}/${unit}.c" "${menu_dir}/${unit}.c" COPYONLY)
        endforeach()
        execute_process(COMMAND patch -s -p1 -i "${CMAKE_SOURCE_DIR}/patches/kyujo2-shared-menu.patch"
            WORKING_DIRECTORY "${menu_dir}" COMMAND_ERROR_IS_FATAL ANY)
        # The MBC3 real-time clock, which GB Tower sets and checks while it saves a Gold or Silver cartridge.
        execute_process(COMMAND patch -s -p1 -i "${CMAKE_SOURCE_DIR}/patches/kyujo2-gbpak-rtc.patch"
            WORKING_DIRECTORY "${menu_dir}" COMMAND_ERROR_IS_FATAL ANY)
        # The Switch SD card's rename() does not replace a file, so settings and Game Boy saves remove it first there.
        execute_process(COMMAND patch -s -p1 -i "${CMAKE_SOURCE_DIR}/patches/kyujo2-switch-replace.patch"
            WORKING_DIRECTORY "${menu_dir}" COMMAND_ERROR_IS_FATAL ANY)
        target_compile_definitions(Kyujo2 PRIVATE KYUJO2_SHARED_MENU)
        target_sources(Kyujo2 PRIVATE native/shared_menu.cpp native/menu_input.c native/windows_cheats.c native/windows_pak.c
            "${menu_dir}/host_menu.c" "${menu_dir}/host_paths.c" "${menu_dir}/host_settings.c"
            "${reference_platform}/host_keys.c" "${reference_platform}/host_input.c"
            "${menu_dir}/host_gbpak.c" "${reference_platform}/host_gbcheck.c" "${reference_platform}/host_gbjp.c"
            "${reference_platform}/../render/png_out.c")
        target_link_options(Kyujo2 PRIVATE -Wl,--wrap=osRecvMesg_recomp)
        if(ANDROID)
            target_compile_definitions(Kyujo2 PRIVATE HOST_SHELL_PICKER)
            target_sources(Kyujo2 PRIVATE native/android_picker.cpp)
        endif()
    endif()
    if(WIN32 AND KYUJO2_REGION STREQUAL "us")
        enable_language(RC)
        if(MINGW)
            set_property(SOURCE native/windows.rc PROPERTY COMPILE_OPTIONS "--preprocessor=${CMAKE_C_COMPILER};--preprocessor-arg=-E;--preprocessor-arg=-xc;--preprocessor-arg=-DRC_INVOKED")
        endif()
        set_property(TARGET Kyujo2 PROPERTY WIN32_EXECUTABLE ON)
        target_compile_definitions(Kyujo2 PRIVATE KYUJO2_WINDOWS_UI NOMINMAX)
        target_sources(Kyujo2 PRIVATE native/menu_windows.cpp native/windows_settings.cpp native/windows.rc)
        target_include_directories(Kyujo2 PRIVATE "${RUNTIME_ROOT}/thirdparty/json")
        target_link_libraries(Kyujo2 PRIVATE comdlg32 comctl32 shell32 ole32)
    endif()
    set(KYUJO_RENDER "${CMAKE_SOURCE_DIR}/source/Kyujo-main/src/render")
    foreach(unit rdp f3dex2 vi_out)
        set(input "${KYUJO_RENDER}/${unit}.c")
        set(change "${CMAKE_SOURCE_DIR}/patches/${unit}-word-order.patch")
        set(output "${CMAKE_CURRENT_BINARY_DIR}/render/${unit}.c")
        add_custom_command(OUTPUT "${output}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/render"
            COMMAND patch --silent -o "${output}" "${input}" "${change}" DEPENDS "${input}" "${change}" VERBATIM)
        target_sources(Kyujo2 PRIVATE "${output}")
        set_property(SOURCE "${output}" APPEND PROPERTY COMPILE_OPTIONS -O2)
    endforeach()
    set(SDL_ROOT "" CACHE PATH "SDL2 platform prefix")
    find_path(SDL_HEADERS SDL.h PATHS "${SDL_ROOT}/include/SDL2" REQUIRED)
    find_library(SDL_LIB SDL2 PATHS "${SDL_ROOT}/lib" "${SDL_ROOT}/lib/x86_64-linux-gnu" REQUIRED)
    target_include_directories(Kyujo2 PRIVATE "${SDL_HEADERS}" "${KYUJO_RENDER}" source/Kyujo-main/src/platform)
    if(WIN32 AND KYUJO2_REGION STREQUAL "us")
        add_executable(Kyujo2WindowsConsoleProbe WIN32 EXCLUDE_FROM_ALL tests/windows_console_probe.c)
        add_executable(Kyujo2WindowsInputTest EXCLUDE_FROM_ALL tests/windows_input.cpp native/windows_input.cpp native/windows_settings.cpp)
        target_include_directories(Kyujo2WindowsInputTest PRIVATE native "${SDL_HEADERS}" "${RUNTIME_ROOT}/thirdparty/json")
        target_compile_definitions(Kyujo2WindowsInputTest PRIVATE NOMINMAX)
        target_link_libraries(Kyujo2WindowsInputTest PRIVATE "${SDL_LIB}")
    endif()
    if(CMAKE_LIBRARY_ARCHITECTURE)
        target_include_directories(Kyujo2 PRIVATE "${SDL_ROOT}/include/${CMAKE_LIBRARY_ARCHITECTURE}")
    endif()
    target_compile_definitions(Kyujo2 PRIVATE RDP_SHELL_NATIVE)
    target_link_libraries(Kyujo2 PRIVATE "${SDL_LIB}")
    # The renderer's GPU back end (OpenGL ES 3): the RDP's drawing runs on the GPU instead of one CPU core. Needed on
    # the Switch, whose CPU cannot draw the game's 3D scenes at full speed. Falls back to the software renderer at run
    # time if no GLES 3 context can be made.
    if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
        set(kyujo2_gpu_default ON)
    else()
        set(kyujo2_gpu_default OFF)
    endif()
    option(KYUJO2_GPU "Draw with the OpenGL ES 3 back end of the renderer" ${kyujo2_gpu_default})
    if(KYUJO2_GPU AND KYUJO2_REGION STREQUAL "us")
        set(gpu_core "${CMAKE_SOURCE_DIR}/source/Kyujo-main/shell/core")
        set(gpu_output "${CMAKE_CURRENT_BINARY_DIR}/render/gpu_gles.c")
        add_custom_command(OUTPUT "${gpu_output}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/render"
            COMMAND patch --silent -o "${gpu_output}" "${gpu_core}/gpu_gles.c" "${CMAKE_SOURCE_DIR}/patches/gpu_gles-kyujo2.patch"
            DEPENDS "${gpu_core}/gpu_gles.c" "${CMAKE_SOURCE_DIR}/patches/gpu_gles-kyujo2.patch" VERBATIM)
        target_sources(Kyujo2 PRIVATE "${gpu_output}" "${gpu_core}/gpu_dispatch.c")
        set_property(SOURCE "${gpu_output}" APPEND PROPERTY COMPILE_OPTIONS -O2)
        target_include_directories(Kyujo2 PRIVATE "${gpu_core}" native)
        target_compile_definitions(Kyujo2 PRIVATE RDP_GPU KYUJO2_GPU)
        if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
            set(switch_portlibs "$ENV{DEVKITPRO}/portlibs/switch")
            if(NOT EXISTS "${switch_portlibs}/include/GLES3/gl3.h")
                set(switch_portlibs /opt/devkitpro/portlibs/switch)
            endif()
            target_include_directories(Kyujo2 PRIVATE "${switch_portlibs}/include")
        else()
            set(GLES_ROOT "" CACHE PATH "Prefix holding include/GLES3 and the GLESv2 library (non-Switch GPU builds)")
            find_path(GLES3_INCLUDE GLES3/gl3.h PATHS "${GLES_ROOT}/include" "${GLES_ROOT}/usr/include" REQUIRED)
            find_library(GLESV2_LIB GLESv2 PATHS "${GLES_ROOT}/lib" "${GLES_ROOT}/usr/lib/x86_64-linux-gnu" REQUIRED)
            target_include_directories(Kyujo2 PRIVATE "${GLES3_INCLUDE}")
            target_link_libraries(Kyujo2 PRIVATE "${GLESV2_LIB}")
        endif()
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
        target_sources(Kyujo2 PRIVATE native/switch_rom.cpp)
        target_compile_options(LiveRecomp PRIVATE -include "${CMAKE_SOURCE_DIR}/native/switch_static_code.h")
        target_link_libraries(Kyujo2 PRIVATE GLESv2 EGL glapi drm_nouveau nx)
        target_link_options(Kyujo2 PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/Kyujo2.map")
        nx_generate_nacp(Kyujo2.nacp NAME "Kyujo2" AUTHOR "Kyujo2" VERSION "${PROJECT_VERSION}")
        nx_create_nro(Kyujo2 NACP Kyujo2.nacp ICON "${CMAKE_SOURCE_DIR}/packaging/switch/icon.jpg")
    endif()
    set_property(SOURCE ${rsp_sources} APPEND PROPERTY COMPILE_OPTIONS -O2)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
        set_property(SOURCE ${rsp_sources} APPEND PROPERTY COMPILE_OPTIONS -msse4.1)
    endif()
    target_link_libraries(Kyujo2 PRIVATE stadium2_game librecomp ultramodern)
    target_compile_definitions(Kyujo2 PRIVATE "KYUJO2_REVISION=\"${KYUJO2_REVISION}\"")
    if(NOT CMAKE_CROSSCOMPILING)
        if(KYUJO2_REGION STREQUAL "us" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
            add_executable(Kyujo2CompletionTest EXCLUDE_FROM_ALL tests/us_completion_wait.c)
            target_link_options(Kyujo2CompletionTest PRIVATE -Wl,--gc-sections)
            target_link_libraries(Kyujo2CompletionTest PRIVATE stadium2_game m)
            add_executable(Kyujo2CupTableTest EXCLUDE_FROM_ALL tests/us_cup_table.c)
            target_link_options(Kyujo2CupTableTest PRIVATE -Wl,--gc-sections)
            target_link_libraries(Kyujo2CupTableTest PRIVATE stadium2_game m)
            add_executable(Kyujo2CartProbeTest EXCLUDE_FROM_ALL tests/us_cart_probes.c)
            target_link_options(Kyujo2CartProbeTest PRIVATE -Wl,--gc-sections)
            target_link_libraries(Kyujo2CartProbeTest PRIVATE stadium2_game m)
            add_executable(Kyujo2MinigameLinkTest EXCLUDE_FROM_ALL tests/us_minigame_link.c)
            target_link_options(Kyujo2MinigameLinkTest PRIVATE -Wl,--gc-sections)
            target_link_libraries(Kyujo2MinigameLinkTest PRIVATE stadium2_game m)
        endif()
        add_executable(Kyujo2OverlayTest EXCLUDE_FROM_ALL tests/overlay_allocations.cpp)
        target_include_directories(Kyujo2OverlayTest PRIVATE "${GUEST_INCLUDE}")
        target_link_libraries(Kyujo2OverlayTest PRIVATE librecomp ultramodern)
    endif()
    if(UNIX AND NOT APPLE)
        target_link_libraries(Kyujo2 PRIVATE atomic)
    endif()
endif()

if(KYUJO2_REGION STREQUAL "us" AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
    add_executable(Kyujo2BattleCheatsTest EXCLUDE_FROM_ALL tests/us_battle_cheats.c native/windows_cheats.c)
    target_include_directories(Kyujo2BattleCheatsTest PRIVATE native "${GUEST_INCLUDE}")
    target_compile_options(Kyujo2BattleCheatsTest PRIVATE -UNDEBUG)
endif()
